// src/outreach/instantlySequenceBuilder.ts

import axios from "axios";
import { logEvent, logError } from "../dashboard/eventsFeed";
import { classifyScraperError } from "../errors/errorService";

/**
 * Instantly API client
 */
const INSTANTLY_API = axios.create({
  baseURL: "https://api.instantly.ai/api/v1",
  headers: {
    "Content-Type": "application/json",
    Authorization: `Bearer ${process.env.INSTANTLY_API_KEY}`
  }
});

/**
 * Create a new Instantly sequence
 */
export async function createInstantlySequence(name: string) {
  logEvent(`Instantly: Creating sequence '${name}'`);

  try {
    const res = await INSTANTLY_API.post("/sequences", {
      name,
      description: "AI‑generated outreach sequence for The Food Advisor"
    });

    const sequenceId = res.data.sequence_id;

    logEvent(`Instantly: Sequence created → ID ${sequenceId}`);

    return sequenceId;

  } catch (err) {
    const category = classifyScraperError(err);
    logError("Instantly: Failed to create sequence", category);
    return null;
  }
}

/**
 * Add an email step to a sequence
 */
export async function addSequenceStep(sequenceId: string, stepNumber: number, subject: string, body: string) {
  logEvent(`Instantly: Adding step ${stepNumber} to sequence ${sequenceId}`);

  try {
    await INSTANTLY_API.post(`/sequences/${sequenceId}/steps`, {
      step_number: stepNumber,
      subject,
      body,
      delay_days: stepNumber === 1 ? 0 : 2 // Email 2 & 3 wait 2 days
    });

    logEvent(`Instantly: Step ${stepNumber} added`);

    return true;

  } catch (err) {
    const category = classifyScraperError(err);
    logError(`Instantly: Failed to add step ${stepNumber}`, category);
    return false;
  }
}

/**
 * Attach a mailbox to the sequence
 */
export async function attachMailbox(sequenceId: string, mailboxId: string) {
  logEvent(`Instantly: Attaching mailbox ${mailboxId} to sequence ${sequenceId}`);

  try {
    await INSTANTLY_API.post(`/sequences/${sequenceId}/mailboxes`, {
      mailbox_id: mailboxId
    });

    logEvent("Instantly: Mailbox attached");
    return true;

  } catch (err) {
    const category = classifyScraperError(err);
    logError("Instantly: Failed to attach mailbox", category);
    return false;
  }
}

/**
 * Build a full outreach sequence (Email 1, 2, 3)
 */
export async function buildFullOutreachSequence(mailboxId: string) {
  logEvent("Instantly: Building full outreach sequence");

  // 1. Create sequence
  const sequenceId = await createInstantlySequence("Food Advisor Outreach");
  if (!sequenceId) return null;

  // 2. Add Email 1
  await addSequenceStep(
    sequenceId,
    1,
    "Quick question about your restaurant",
    "Hi {{restaurant_name}},\n\nI run The Food Advisor — a curated directory of great restaurants.\n\nWe’d love to feature you. Can I send more details?"
  );

  // 3. Add Email 2
  await addSequenceStep(
    sequenceId,
    2,
    "Following up — The Food Advisor",
    "Hi again {{restaurant_name}},\n\nJust checking in — we’d still love to include you.\n\nWould you like more info?"
  );

  // 4. Add Email 3
  await addSequenceStep(
    sequenceId,
    3,
    "Final follow‑up",
    "Hi {{restaurant_name}},\n\nThis is my last message — happy to send details if you're interested.\n\nThanks!"
  );

  // 5. Attach mailbox
  await attachMailbox(sequenceId, mailboxId);

  logEvent(`Instantly: Full outreach sequence ready → ID ${sequenceId}`);

  return sequenceId;
}
