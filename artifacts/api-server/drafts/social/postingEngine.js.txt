export async function postAutomatically(post) {
  const results = [];

  for (const platform of post.platforms) {
    if (platform === "instagram") {
      results.push(await postToInstagram(post));
    }
    if (platform === "facebook") {
      results.push(await postToFacebook(post));
    }
    if (platform === "tiktok") {
      results.push(await postToTikTok(post));
    }
    if (platform === "x") {
      results.push(await postToX(post));
    }
    if (platform === "google") {
      results.push(await postToGoogleBusiness(post));
    }
  }

  return results;
}
