import { getScheduledPosts } from "./scheduler.js";
import { postAutomatically } from "./postingEngine.js";

export async function runSocialMediaCron() {
  const posts = await getScheduledPosts();

  for (const post of posts) {
    const now = Date.now();
    const scheduled = new Date(post.scheduledFor).getTime();

    if (scheduled <= now) {
      await postAutomatically(post);
    }
  }
}
